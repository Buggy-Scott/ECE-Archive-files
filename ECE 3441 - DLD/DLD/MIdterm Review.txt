++i (pre increment) - is the first increment, meaning it will add a new value to a list.
// Pre-increment
    printf("++i: %d\n", ++i);  // Output: 6 (incremented first)
    printf("After ++i: i = %d\n", i);  // i is now 6
i++ (post increment) - allows the current value to increment to itself.
// Post-increment  
    printf("j++: %d\n", j++);  // Output: 5 (original value used first)
    printf("After j++: j = %d\n", j);  // j is now 6

 void rString(char str[]){
    if(str == NULL) return;

    int leng = strlen(str);
    int start =0;
    int end = leng -1;

    while(start < end){
        char temp = str[start];
        str[start] = str[end];
        str[end] = temp;

        start++;
        end--;
    }
 }
